# Core: transcription and media fetching.

# YouTube PO-token plugin, paired with the bgutil.service HTTP server on
# this same droplet (see BGUTIL_POT_BASE_URL in basiq_agent.py). The plugin
# refuses to run at all if its major version doesn't match the server's --
# and yt-dlp getting force-upgraded weekly by basiq-ytdlp-update.service
# while this line was absent from requirements.txt meant this package sat
# untouched at a stale major version until it broke every YouTube grab with
# "Plugin and script major versions are mismatched" (confirmed 2026-09-29:
# droplet had 1.3.1 against the server's 2.0.0, sat that way silently since
# the server was manually bumped weeks earlier). Bump this line -- and
# re-run the update service or `systemctl restart basiq-agent` -- any time
# bgutil.service itself is upgraded to a new major version.
bgutil-ytdlp-pot-provider==2.0.0

# Live-capture transcript streaming: feeds a second, audio-only ffmpeg

websocket-client>=1.7.0

# Generic live-stream resolution: yt-dlp only knows sites with a dedicated
# extractor. For everything else (e.g. CBS News, ABC News -- major network
# live pages with no yt-dlp extractor at all), a headless browser loads the
# page and watches its own network traffic for the manifest URL the site's
# own player uses. Degrades gracefully -- if this isn't installed, live
# capture just falls back to yt-dlp-only coverage.
# After installing, also run: playwright install firefox
# (basiq_agent.py's generic live resolver uses Firefox, not Chromium, as of
# 2026-09-24 -- Akamai Bot Manager challenged automated Chromium on a real
# login-gated source. `playwright install chromium` alone is no longer
# sufficient for that path.)
playwright>=1.40.0
